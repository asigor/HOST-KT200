def get_timestamp():
    """Generuje timestamp w formacie używanym przez KT200II"""
    now = datetime.now()
    return now.strftime(f"%Y-%m-%d %H:%M:%S (UTC{TIME_ZONE_OFFSET})")

def log_request_response(endpoint, request_data, response_data, time_offset=0):
    """
    Loguje żądanie i odpowiedź w formacie szablonu.
    
    Args:
        endpoint: Ścieżka endpointu
        request_data: Dane żądania
        response_data: Dane odpowiedzi
        time_offset: Przesunięcie czasu w sekundach dla odpowiedzi
    """
    now = datetime.now()
    request_timestamp = now.strftime(f"%Y-%m-%d %H:%M:%S (UTC{TIME_ZONE_OFFSET})")
    response_timestamp = (now + timedelta(seconds=time_offset)).strftime(f"%Y-%m-%d %H:%M:%S (UTC{TIME_ZONE_OFFSET})")
    
    # Formatowanie logu
    log_entry = f"""[{request_timestamp}] [Request]
Address: http://{HOST}{endpoint}?{request_data}

[{response_timestamp}] [Response]
Body: {response_data}
"""
    
    # Zapisywanie logu do pliku
    with open(LOG_FILE, "a", encoding="utf-8") as f:
        f.write(log_entry + "\n")
    
    if DEBUG:
        print(log_entry)
    
    return request_timestamp, response_timestamp

def encrypt_with_xor(data, key=XOR_KEY):
    """
    Szyfruje dane używając XOR z podanym kluczem.
    
    Args:
        data: Dane do zaszyfrowania
        key: Klucz XOR (domyślnie 85)
        
    Returns:
        Zaszyfrowane dane w formacie base64
    """
    if isinstance(data, str):
        data = data.encode('utf-8')
    
    encrypted_bytes = bytes([b ^ key for b in data])
    return base64.b64encode(encrypted_bytes).decode('utf-8')

def decrypt_with_xor(base64_data, key=XOR_KEY):
    """
    Deszyfruje dane używając XOR z podanym kluczem.
    
    Args:
        base64_data: Dane zaszyfrowane w formacie base64
        key: Klucz XOR (domyślnie 85)
        
    Returns:
        Odszyfrowane dane
    """
    try:
        encrypted_bytes = base64.b64decode(base64_data)
        decrypted_bytes = bytes([b ^ key for b in encrypted_bytes])
        return decrypted_bytes.decode('utf-8')
    except:
        return ""

def random_base64(length=300):
    """
    Generuje losowy ciąg w formacie Base64 o podanej długości.
    
    Args:
        length: Długość danych binarnych przed kodowaniem (domyślnie 300)
        
    Returns:
        Losowy ciąg w formacie Base64
    """
    random_bytes = os.urandom(length)
    return base64.b64encode(random_bytes).decode('utf-8')

def random_base64_url_safe(length=300):
    """
    Generuje losowy ciąg w formacie Base64 URL-safe.
    
    Args:
        length: Długość danych binarnych przed kodowaniem (domyślnie 300)
        
    Returns:
        Losowy ciąg w formacie Base64 URL-safe
    """
    random_bytes = os.urandom(length)
    base64_data = base64.b64encode(random_bytes).decode('utf-8')
    return base64_data.replace('+', '-').replace('/', '_').replace('=', '')

def generate_sha1(data=None):
    """
    Generuje hash SHA-1 dla podanych danych lub danych losowych.
    
    Args:
        data: Dane do zahashowania (domyślnie None, co generuje losowe dane)
        
    Returns:
        Hash SHA-1 w formacie hex
    """
    if data is None:
        data = os.urandom(20)
    
    if isinstance(data, str):
        data = data.encode('utf-8')
    
    return hashlib.sha1(data).hexdigest()

def generate_activation_response(code, hwid, hash_value, stage=1):
    """
    Generuje odpowiedź dla endpointu activation.php.
    
    Proces:
    1. Weryfikuje hash (opcjonalne)
    2. Łączy hwid + code jako payload
    3. Generuje HMAC-SHA1 używając tajnego klucza
    4. Tworzy blok base64 z HMAC + losowymi danymi
    
    Args:
        code: Kod aktywacyjny
        hwid: Identyfikator sprzętowy
        hash_value: Hash przesłany w żądaniu
        stage: Etap aktywacji (1 lub 2)
        
    Returns:
        Odpowiedź serwera
    """
    # Dekodowanie hwid z URL
    hwid = hwid.replace('%2b', '+').replace('%2f', '/').replace('%3d', '=')
    
    # Tworzenie payloadu (hwid + code)
    payload = (hwid + code).encode('utf-8')
    
    # Generowanie HMAC-SHA1
    signature = hmac.new(HMAC_SECRET_KEY, payload, hashlib.sha1).digest()
    signature_b64 = base64.b64encode(signature).decode('utf-8')
    
    # Generowanie losowych danych do wypełnienia odpowiedzi
    random_data = random_base64(400 if stage == 1 else 450)
    
    # Tworzenie odpowiedzi w formacie: "OK\n[base64_data]"
    return f"OK\n{random_data}"

def verify_activation_hash(code, hwid, hash_value):
    """
    Weryfikuje hash przesłany w żądaniu activation.php.
    
    Proces:
    1. Dekoduje hwid z Base64 do hex
    2. Łączy code|hwid_hex|timestamp jako payload
    3. Generuje SHA-1 i koduje do Base64
    4. Porównuje z przesłanym hashem
    
    Args:
        code: Kod aktywacyjny
        hwid: Identyfikator sprzętowy (Base64)
        hash_value: Hash przesłany w żądaniu
        
    Returns:
        True jeśli hash jest poprawny, False w przeciwnym razie
    """
    # Jest to uproszczona implementacja - rzeczywista weryfikacja byłaby bardziej złożona
    try:
        # Dekodowanie hwid z URL i konwersja na hex
        hwid = hwid.replace('%2b', '+').replace('%2f', '/').replace('%3d', '=')
        hwid_bytes = base64.b64decode(hwid)
        hwid_hex = hwid_bytes.hex()
        
        # Utworzenie payload dla różnych znaczników czasu
        # W rzeczywistości, timestamp mógłby być generowany na podstawie innych danych
        for offset in range(-60, 60, 5):  # Sprawdza różne timestampy w zakresie +/- 60 sekund
            ts = str(int(time.time()) + offset)
            payload = f"{code}|{hwid_hex}|{ts}"
            
            # Generowanie hash SHA-1 i kodowanie do Base64
            calculated_hash = base64.b64encode(
                hashlib.sha1(payload.encode('utf-8')).digest()
            ).decode('ascii')
            
            # Porównanie z przesłanym hashem
            if calculated_hash == hash_value.replace('%3d', '='):
                return True
        
        return False
    except:
        return False  # W przypadku błędu zakładamy, że hash jest niepoprawny

def generate_init2024_response(name, hwid, length):
    """
    Generuje odpowiedź dla endpointu init2024.php.
    
    Proces:
    1. Generuje losowy token (5-10 znaków)
    2. Generuje losowe dane base64 (10-20 bajtów)
    3. Łączy token + dane + "||" + liczba
    
    Args:
        name: Parametr name z żądania
        hwid: Identyfikator sprzętowy
        length: Długość podana w żądaniu
        
    Returns:
        Odpowiedź w formacie "[token][random_base64]||[number]"
    """
    # Generowanie losowego tokenu
    token_length = random.randint(5, 10)
    token = ''.join(random.choices('abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-_', k=token_length))
    
    # Generowanie losowych danych base64
    random_data_length = random.randint(10, 20)
    random_data = base64.b64encode(os.urandom(random_data_length)).decode('ascii')
    random_data = random_data.replace('+', '-').replace('/', '_').replace('=', '')
    
    # Generowanie liczby na końcu (zwykle 2 cyfry)
    number = random.randint(10, 99)
    
    # Tworzenie dodatkowych danych ukrytych w odpowiedzi
    # W rzeczywistym serwerze tutaj mogą być przekazywane informacje konfiguracyjne
    timestamp = int(time.time())
    hidden_data = f"{timestamp:016d}"  # 16-cyfrowy timestamp
    
    # Łączenie wszystkich elementów
    response = f"{token}{random_data}{hidden_data}||{number}"
    
    return response

@app.route('/generate-honey')
def generate_honey():
    """Generuje plik Honey.bin z logów komunikacji"""
    if not os.path.exists(LOG_FILE):
        return "Błąd: Brak pliku komunikacja_log.txt!"
    
    try:
        with open(LOG_FILE, "r", encoding="utf-8") as f:
            log_data = f.read()
        
        # Szyfrowanie logów używając XOR key=85
        encrypted_lines = [encrypt_with_xor(line) for line in log_data.strip().split('\n')]
        
        # Zapisanie zaszyfrowanych danych do Honey.bin
        with open(HONEY_BIN_FILE, "w", encoding="utf-8") as f:
            f.write('\n'.join(encrypted_lines))
        
        return redirect(url_for('index'))
    except Exception as e:
        return f"Błąd podczas generowania Honey.bin: {str(e)}"

@app.route('/generate-template')
def generate_template():
    """Generuje wypełniony szablon KT200II_exe.txt"""
    if not os.path.exists(LOG_FILE):
        return "Błąd: Brak pliku komunikacja_log.txt!"
    
    try:
        with open(LOG_FILE, "r", encoding="utf-8") as f:
            log_data = f.read()
        
        # Analiza logów w celu wyodrębnienia danych do szablonu
        data = {}
        
        # Regularne wyrażenia do wyodrębniania danych
        activation1_re = r"Address: http://.*?/weblm/activation\.php\?code=([^&]+)&hwid=([^&]+)&hash=([^&\n]+)"
        handshake_re = r"Address: http://.*?/handshake\.php\?id=([^&]+)&len=([^&]+)&name=([^&]+)&vers=([^&]+)&type=([^&\n]+)"
        init2024_re = r"Address: http://.*?/init2024\.php\?name=([^&]+)&id=([^&]+)&len=([^&\n]+)"
        activation2_re = r"Address: http://.*?/weblm/activation\.php\?code=([^&]+)&hwid=([^&]+)&hash=([^&\n]+)"
        chatgpt_re = r"Address: http://.*?/chatgpt\.php\?name=([^&]+)&tag=([^&\n]+)"
        comm2024_re = r"Address: http://.*?/comm2024\.php\?name=([^&]+)&tick=([^&\n]+)"
        
        # Wyodrębnienie adresu hosta
        host_match = re.search(r"Address: http://([^/]+)/", log_data)
        if host_match:
            data["HOST"] = host_match.group(1)
        
        # Wyodrębnienie danych z pierwszego żądania aktywacji
        activation1_match = re.search(activation1_re, log_data)
        if activation1_match:
            data["CODE1"] = activation1_match.group(1)
            data["HWID"] = activation1_match.group(2)
            data["HASH1"] = activation1_match.group(3)
            
            # Wyodrębnienie odpowiedzi
            activation1_response_re = r"\[Response\]\nBody: (OK\n.*?)(?=\[|$)"
            activation1_response_match = re.search(activation1_response_re, log_data, re.DOTALL)
            if activation1_response_match:
                data["RESPONSE1"] = activation1_response_match.group(1).strip()
        
        # Wyodrębnienie danych z żądania handshake
        handshake_match = re.search(handshake_re, log_data)
        if handshake_match:
            data["LEN"] = handshake_match.group(2)
            data["ENCODEDNAME"] = handshake_match.group(3)
            data["VERSION"] = handshake_match.group(4)
            data["TYPE"] = handshake_match.group(5)
            
            # Wyodrębnienie odpowiedzi
            handshake_response_re = r"handshake\.php.*?\n\[.*?\] \[Response\]\nBody: (.*?)(?=\[|$)"
            handshake_response_match = re.search(handshake_response_re, log_data, re.DOTALL)
            if handshake_response_match:
                data["RESPONSE2"] = handshake_response_match.group(1).strip()
        
        # Wyodrębnienie danych z żądania init2024
        init2024_match = re.search(init2024_re, log_data)
        if init2024_match:
            data["NAME1"] = init2024_match.group(1)
            
            # Wyodrębnienie odpowiedzi
            init2024_response_re = r"init2024\.php.*?\n\[.*?\] \[Response\]\nBody: (.*?)(?=\[|$)"
            init2024_response_match = re.search(init2024_response_re, log_data, re.DOTALL)
            if init2024_response_match:
                data["RESPONSE3"] = init2024_response_match.group(1).strip()
        
        # Wyodrębnienie danych z drugiego żądania aktywacji
        # Pomijamy pierwsze dopasowanie (które już przetwarzaliśmy)
        activation_matches = re.finditer(activation1_re, log_data)
        next(activation_matches, None)  # Pomijamy pierwsze dopasowanie
        activation2_match = next(activation_matches, None)
        if activation2_match:
            data["CODE2"] = activation2_match.group(1)
            data["HASH2"] = activation2_match.group(3)
            
            # Wyodrębnienie odpowiedzi
            activation_response_matches = re.finditer(r"\[Response\]\nBody: (OK\n.*?)(?=\[|$)", log_data, re.DOTALL)
            next(activation_response_matches, None)  # Pomijamy pierwszą odpowiedź
            activation2_response_match = next(activation_response_matches, None)
            if activation2_response_match:
                data["RESPONSE4"] = activation2_response_match.group(1).strip()
        
        # Wyodrębnienie danych z żądania chatgpt
        chatgpt_match = re.search(chatgpt_re, log_data)
        if chatgpt_match:
            data["GPTNAME"] = chatgpt_match.group(1)
            data["TAG"] = chatgpt_match.group(2)
            
            # Wyodrębnienie odpowiedzi
            chatgpt_response_re = r"chatgpt\.php.*?\n\[.*?\] \[Response\]\nBody: (.*?)(?=\[|$)"
            chatgpt_response_match = re.search(chatgpt_response_re, log_data, re.DOTALL)
            if chatgpt_response_match:
                data["GPT_RESPONSE"] = chatgpt_response_match.group(1).strip()
        
        # Wyodrębnienie danych z żądania comm2024
        comm2024_match = re.search(comm2024_re, log_data)
        if comm2024_match:
            data["TICK"] = comm2024_match.group(2)
            
            # Wyodrębnienie odpowiedzi
            comm2024_response_re = r"comm2024\.php.*?\n\[.*?\] \[Response\]\nBody: (.*?)(?=\[|$)"
            comm2024_response_match = re.search(comm2024_response_re, log_data, re.DOTALL)
            if comm2024_response_match:
                data["COMM_RESPONSE"] = comm2024_response_match.group(1).strip()
        
        # Wyodrębnienie znaczników czasu
        timestamps = re.findall(r"\[(.*?)\] \[Request\]", log_data)
        response_timestamps = re.findall(r"\[(.*?)\] \[Response\]", log_data)
        
        if len(timestamps) >= 6 and len(response_timestamps) >= 6:
            data["TIME_ACTIVATION1"] = timestamps[0]
            data["TIME_RESPONSE1"] = response_timestamps[0]
            data["TIME_HANDSHAKE"] = timestamps[1]
            data["TIME_RESPONSE2"] = response_timestamps[1]
            data["TIME_INIT"] = timestamps[2]
            data["TIME_RESPONSE3"] = response_timestamps[2]
            data["TIME_ACTIVATION2"] = timestamps[3]
            data["TIME_RESPONSE4"] = response_timestamps[3]
            data["TIME_CHATGPT"] = timestamps[4]
            data["TIME_RESPONSE5"] = response_timestamps[4]
            data["TIME_COMM2024"] = timestamps[5]
            data["TIME_RESPONSE6"] = response_timestamps[5]
        
        # Wczytanie szablonu
        template = """[{{TIME_ACTIVATION1}}] [Request]
Address: http://{{HOST}}/weblm/activation.php?code={{CODE1}}&hwid={{HWID}}&hash={{HASH1}}

[{{TIME_RESPONSE1}}] [Response]
Body: {{RESPONSE1}}

[{{TIME_HANDSHAKE}}] [Request]
Address: http://{{HOST}}/handshake.php?id={{HWID}}&len={{LEN}}&name={{ENCODEDNAME}}&vers={{VERSION}}&type={{TYPE}}

[{{TIME_RESPONSE2}}] [Response]
Body: {{RESPONSE2}}

[{{TIME_INIT}}] [Request]
Address: http://{{HOST}}/init2024.php?name={{NAME1}}&id={{HWID}}&len={{LEN}}

[{{TIME_RESPONSE3}}] [Response]
Body: {{RESPONSE3}}

[{{TIME_ACTIVATION2}}] [Request]
Address: http://{{HOST}}/weblm/activation.php?code={{CODE2}}&hwid={{HWID}}&hash={{HASH2}}

[{{TIME_RESPONSE4}}] [Response]
Body: {{RESPONSE4}}

[{{TIME_CHATGPT}}] [Request]
Address: http://{{HOST}}/chatgpt.php?name={{GPTNAME}}&tag={{TAG}}

[{{TIME_RESPONSE5}}] [Response]
Body: {{GPT_RESPONSE}}

[{{TIME_COMM2024}}] [Request]
Address: http://{{HOST}}/comm2024.php?name={{HWID}}&tick={{TICK}}

[{{TIME_RESPONSE6}}] [Response]
Body: {{COMM_RESPONSE}}"""
        
        # Wypełnienie szablonu danymi
        for key, value in data.items():
            template = template.replace(f"{{{{{key}}}}}", value)
        
        # Zapisanie wypełnionego szablonu
        with open("WYPELNIONY_KT200II_exe.txt", "w", encoding="utf-8") as f:
            f.write(template)

    except Exception as e:
        return f"Błąd podczas generowania szablonu: {str(e)}"
